# 1. Создаем  виртуальное окружение Python 3.14
python -m venv venv

# 2. Активируем его и разрешаем запуск скриптов на Windows
(Set-ExecutionPolicy -Scope Process -ExecutionPolicy RemoteSigned) ; (& ".\venv\Scripts\Activate.ps1")

# 3. Устанавливаем Django и библиотеки для работы с Excel таблицами 
pip install django openpyxl

# 4. Создаем чистый Django проект с именем core в папке
django-admin startproject core .

# 5. Создаем приложение helpdesk 
python manage.py startapp HelpDesk

# 6. Сканируем файл моделей и создаем таблицы бд
python manage.py makemigrations HelpDesk

# 7. применяем изменения и строим таблицы внутри файла db.sqlite3
python manage.py migrate

# 8.Создаем админа
python manage.py createsuperuser 

# 9. Создаем новую изолированную ветку для текущей техподдержки
git checkout -b feature-HelpDesk-system

# 10. Выбираем абсолютно все отлаженные файлы (модели, сервисы, админку, HTML)
git add .

# 11. Делаем строгий фиксирующий коммит (все комментарии и исправления на месте!)
git commit -m "feat: complete helpdesk support system with excel importer, inline comments and manual save fixes"

# 12. Принудительно выталкиваем именно текущую новую ветку на GitHub
git push origin feature-HelpDesk-system --force
